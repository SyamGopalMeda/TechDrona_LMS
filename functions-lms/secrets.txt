# Secret Manager secrets mounted into every LMS function, under the same environment variable
# names as the VM's /opt/techdrona/server.env. Values are NEVER stored in this repo:
#   firebase functions:secrets:set <NAME> --project billing-f4fe3
# A name listed here must exist before `firebase deploy` (the deploy fails otherwise).
TD_SERVICE_TOKEN           # edge -> API service token (also keys the pagination cursor HMAC)
TD_CSRF_KEY                # CSRF HMAC key (same value as the VM web container keeps tokens valid)
TD_K_DEVICE_KEY_V1         # device-cookie HMAC keyring
TD_K_LOGIN_KEY_V1          # login identifier HMAC keyring
TD_K_TOTP_KEY_V1           # TOTP secret encryption keyring
TD_FP_HMAC_KEY
TD_LOCAL_ENVELOPE_KEY
TD_ANALYTICS_SUBJECT_KEY
TD_K_TESTCASE_KEY_V1       # coding hidden-test-case key wrapping (same value as the VM)
RAZORPAY_WEBHOOK_SECRET    # payment webhook signature (same value as the VM)
AWS_ACCESS_KEY_ID          # Cloud Storage HMAC access id (S3 interoperability)
AWS_SECRET_ACCESS_KEY      # Cloud Storage HMAC secret
TD_COLLEGE_PORTAL_SECRET   # /college-portal proxy -> Cloud Run college-portal (its PORTAL_PROXY_SECRET)
# DATABASE_URL: none. Owner decision 2026-11-12: the Firebase stack runs without Postgres.
